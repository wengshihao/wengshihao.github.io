  <script>
    // Move the dot along the blue path from the moment the chart is parsed.
    // (SVG <animateMotion> would wait for the whole page, fonts and photo
    // included, to finish loading before it starts.)
    (function () {
      var path = document.getElementById('vm-vis');
      var dot = document.querySelector('.hp-vm__chart .vm-dot');
      if (!path || !dot || !path.getTotalLength) return;
      if (window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches) return;
      var len = path.getTotalLength(), period = 5000, travel = .7, t0 = null;
      function frame(t) {
        if (t0 === null) t0 = t;
        var k = ((t - t0) % period) / period;
        var p = path.getPointAtLength(len * Math.min(k / travel, 1));
        dot.setAttribute('cx', p.x.toFixed(2));
        dot.setAttribute('cy', p.y.toFixed(2));
        requestAnimationFrame(frame);
      }
      requestAnimationFrame(frame);
    })();
  </script>
